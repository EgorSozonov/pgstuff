//Install (package names may differ)
pkInstall postgresql postgresql-client
sudo -u postgres psql -p 5589 -c "SELECT version();"
sudo mkdir -p /usr/local/pg/data
sudo chown postgres /usr/local/pg /usr/local/pg/data
sudo mkdir -p /var/local/log/pg
sudo chown postgres /var/local/log/pg

//Initialize the cluster
sudo -u postgres initdb -D /usr/local/pg/data --locale-provider builtin -E UTF-8 --locale=C.UTF-8

//Prepare to run & run
sudo mkdir -p /run/postgresql
sudo chown postgres /run/postgresql
sudo -u postgres /usr/bin/pg_ctl -l /var/local/log/pg/log -D /usr/local/pg/data -o '-p 5589' start
